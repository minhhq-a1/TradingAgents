safeDiv(float num, float den, float fallback = 0.0) =>
    den != 0 and not na(num) and not na(den) ? num / den : fallback

fmtPrice(float v) =>
    na(v) ? "—" : str.tostring(v, format.mintick)

fmtScore(float v) =>
    na(v) ? "—" : str.tostring(math.round(v))

formatPctFromEntry(float level, float entry) =>
    string res = ""
    if showPctOnLabelsInput and not na(level) and not na(entry) and entry != 0
        float pct   = (level - entry) / entry * 100.0
        string sign = pct >= 0 ? "+" : ""
        res := " (" + sign + str.tostring(pct, "#.##") + "%)"
    res

sizeFromString(string s) =>
    s == "Tiny" ? size.tiny : s == "Small" ? size.small : s == "Normal" ? size.normal : s == "Large" ? size.large : s == "Huge" ? size.huge : size.small

sizeDownFromString(string s) =>
    s == "Tiny" ? size.tiny : s == "Small" ? size.tiny : s == "Normal" ? size.small : s == "Large" ? size.normal : s == "Huge" ? size.large : size.tiny

lineStyleFromString(string s) =>
    s == "Dashed" ? line.style_dashed : s == "Dotted" ? line.style_dotted : line.style_solid

clamp01(float x) =>
    math.max(0.0, math.min(1.0, x))

buildGauge(float value, float maxVal, int width) =>
    string result = ""
    int filled = maxVal <= 0 ? 0 : math.round(math.min(width, math.max(0.0, value / maxVal * width)))
    for i = 0 to width - 1
        result += i < filled ? "▰" : "▱"
    result

periodStats(array<int> tms, array<bool> wns, int cutoff) =>
    int w = 0
    int l = 0
    int n = array.size(tms)
    if n > 0
        for i = 0 to n - 1
            if array.get(tms, i) >= cutoff
                if array.get(wns, i)
                    w += 1
                else
                    l += 1
    [w, l]

// statistics of the leg covering bars (startBar, endBar]
legStats(int startBar, int endBar) =>
    float s  = 0.0
    float pk = 0.0
    int   n  = 0
    int   sz = array.size(effBuf)
    if endBar > startBar
        for k = startBar + 1 to endBar
            int idx = sz - 1 - (bar_index - k)
            if idx >= 0 and idx < sz
                float v = array.get(effBuf, idx)
                s  += v
                pk := math.max(pk, v)
                n  += 1
    [n > 0 ? s / n : 1.0, pk, n]

bool  newSwing = false

if confirmed and isWarmedUp
    if legDir == 0
        legDir        := 1
        legStartBar   := bar_index
        legStartPrice := low
        extPrice      := high
        extBar        := bar_index
    else if legDir == 1
        if high > extPrice
            extPrice := high
            extBar   := bar_index
        if close < extPrice - swingRevInput * atr
            [le, lp, ln] = legStats(legStartBar, extBar)
            sw := Swing.new(1, extPrice, extBar, le, lp, safeDiv(extPrice - legStartPrice, atr, 0.0), ln, bar_index)
            newSwing := true
            legDir        := -1
            legStartBar   := extBar
            legStartPrice := extPrice
            if extBar == bar_index
                extPrice := low
                extBar   := bar_index
            else
                extPrice := lowSinceExt
                extBar   := bar_index + lowSinceOff
    else
        if low < extPrice
            extPrice := low
            extBar   := bar_index
        if close > extPrice + swingRevInput * atr
            [le, lp, ln] = legStats(legStartBar, extBar)
            sw := Swing.new(-1, extPrice, extBar, le, lp, safeDiv(legStartPrice - extPrice, atr, 0.0), ln, bar_index)
            newSwing := true
            legDir        := 1
            legStartBar   := extBar
            legStartPrice := extPrice
            if extBar == bar_index
                extPrice := high
                extBar   := bar_index
            else
                extPrice := highSinceExt
                extBar   := bar_index + highSinceOff
    if newSwing
        array.push(swings, sw)
        if array.size(swings) > MAX_SWINGS
            array.shift(swings)
        if showSwingsInput
            label.new(sw.bar, sw.price, "", xloc = xloc.bar_index, style = label.style_circle, color = SWING_DOT, size = size.tiny)

// Climactic run ending at the newest swing (kind = -1 for a selling climax, +1 for a buying climax):
// walk back while the opposite-kind swings keep falling (rising) and the same-kind swings stay above (below)
// the candidate. Returns [runStartBar, runStartPrice, sameKindSwingsInsideRun].
runOf(int kind, float candPx) =>
    int   n        = array.size(swings)
    int   startBar = na
    float startPx  = na
    int   cnt      = 0
    if n >= 2
        int   i = n - 2
        Swing s = array.get(swings, i)
        startBar := s.bar
        startPx  := s.price
        bool go = true
        while go and i - 2 >= 0
            Swing prevOpp  = array.get(swings, i - 2)
            Swing between  = array.get(swings, i - 1)
            bool oppOk     = kind == -1 ? prevOpp.price > s.price : prevOpp.price < s.price
            bool betweenOk = kind == -1 ? between.price > candPx : between.price < candPx
            if oppOk and betweenOk
                i -= 2
                s := prevOpp
                startBar := s.bar
                startPx  := s.price
                cnt += 1
            else
                go := false
    [startBar, startPx, cnt]

evLabel(Rng r, int x, float y, string txt, bool atClimaxSide, color c, string sz, string tip = "") =>

        label mk = schemLabel(x, yy, txt, c, sz, tip)

climEdge(Rng r) =>
    r.side == 1 ? r.bot : r.top

reactEdge(Rng r) =>
    r.side == 1 ? r.top : r.bot

redrawRange(Rng r) =>

    line.set_y2(r.edgeReact, reactEdge(r))
    label.set_y(r.tagClim, climEdge(r))
    label.set_y(r.tagReact, reactEdge(r))
    label.set_y(r.stageLbl, reactEdge(r))
    true

// vertical dashed divider at x closing the phase `ended`, with its caption centered on the closed phase
addDivider(Rng r, int x, int ended, color c) =>
    float h   = math.max(r.top - r.bot, syminfo.mintick)
    float yTop = r.top + DIVIDER_PAD_FRAC * h
    float yBot = r.bot - (PHASE_TAG_FRAC + DIVIDER_PAD_FRAC) * h
    if showPhasesInput and x > r.lastDivBar
        array.push(r.dividers, line.new(x, yTop, x, yBot, xloc = xloc.bar_index, color = color.new(inkOf(c), 15), style = line.style_dashed, width = 1))
        if x - r.lastDivBar >= MIN_PHASE_BARS
            int mid = math.round((r.lastDivBar + x) / 2.0)
            array.push(r.phaseTags, schemLabel(mid, r.bot - PHASE_TAG_FRAC * h, "Phase " + phaseTxt(ended), c, eventSize))
        r.lastDivBar := x
    true

removeOutcome(int id) =>
    n = 0
    if array.size(oId) > 0
        for i = array.size(oId) - 1 to 0
            if array.get(oId, i) == id
                n += 1
                array.remove(oId, i)
                array.remove(oDir, i)
                array.remove(oTgt, i)
                array.remove(oInv, i)
                array.remove(oBar, i)
    n

// finalization flags for this bar
bool   closeRange = false
bool   failedRange = false
string closeTag    = ""

// ── A) bar-level checks on the active range ──
if confirmed and not na(cur)
    float h    = math.max(cur.top - cur.bot, syminfo.mintick)
    float uC   = uOf(cur, close)
    color rngCol = cur.side == 1 ? bullColorInput : bearColorInput

    // 1) expiry (before events)
    if not cur.brokeOut and bar_index - cur.climaxBar > maxRangeBarsInput
        closeRange := true
        failedRange := true
        closeTag := "TIMED OUT"

    // 2) decisive break beyond the climax edge
    if not closeRange and not cur.brokeOut and uC < -failDistAtrInput * atr / h
        evLabel(cur, bar_index, cur.side == 1 ? low : high, nmOf(cur, "BREAKDOWN", "BREAKOUT"), true, failColorInput, eventSize, "Decisive close beyond the climax edge: range failed")
        if cur.side == 1
            accBad += 1
        else
            distBad += 1
        evFailed := true
        eventMsg += nmOf(cur, "Accumulation", "Distribution") + " range FAILED (decisive break) | "
        closeRange := true
        failedRange := true
        closeTag := "BROKEN"

    // 3) pending spring: reclaim or timeout
    if not closeRange and cur.pendBreak
        if uC >= 0
            cur.pendBreak := false
            if requireSTInput and cur.nTests == 0
                if na(cur.innerClim)
                    cur.innerClim := line.new(cur.climaxBar, climEdge(cur), bar_index, climEdge(cur), xloc = xloc.bar_index, color = color.new(inkOf(rngCol), 35), width = 1)
                if cur.side == 1
                    cur.bot := cur.pendPrice
                else
                    cur.top := cur.pendPrice
                redrawRange(cur)
                evLabel(cur, cur.pendBar, cur.pendPrice, "ext", true, climColorInput, eventSizeSub, "Climax extreme extended before the first ST")
            else
                cur.sprung  := true
                cur.tested    := false
                cur.springPrice := cur.pendPrice
                cur.springBar   := cur.pendBar
                cur.springEff   := cur.pendEff
                cur.stage       := 3
                float depthAtr = safeDiv(cur.side == 1 ? cur.bot - cur.pendPrice : cur.pendPrice - cur.top, atr, 0.0)
                float depthC = clamp01(1.0 - depthAtr / 1.5)
                float effC   = clamp01(1.0 - safeDiv(cur.pendEff, cur.climEff, 1.0))
                float speedC = clamp01(1.0 - (bar_index - cur.pendConfBar) / 10.0)
                cur.springScore := math.round(40.0 * depthC + 40.0 * effC + 20.0 * speedC)
                string effNote = cur.pendEff < 1.0 ? nmOf(cur, "no supply", "no demand") : "shake-out"
                string sprDet = str.tostring(cur.springScore) + "/100 · " + effNote + " · " + str.tostring(depthAtr, "#.##") + " ATR"
                evLabel(cur, cur.pendBar, cur.pendPrice, nmOf(cur, "Spring", "UTAD") + (detailLblInput ? "\n" + sprDet : ""), true, sigColorInput, eventSize, nmOf(cur, "Spring", "Upthrust After Distribution") + "\n" + sprDet)
                addDivider(cur, cur.pendBar, 2, rngCol)
                sigDetect := true
                eventLevel := cur.pendPrice
                entryScoreEv := cur.springScore
                entryInfoEv := nmOf(cur, "Spring", "UTAD") + " @ " + fmtPrice(cur.pendPrice) + " (" + effNote + ")"
                if cur.side == 1
                    evSpring := true
                    if not cur.springCounted
                        springRanges += 1
                        cur.springCounted := true
                else
                    evUtad := true
                    if not cur.springCounted
                        utadRanges += 1
                        cur.springCounted := true
                eventMsg += nmOf(cur, "SPRING", "UTAD") + " " + str.tostring(cur.springScore) + "/100 (" + effNote + ") at " + fmtPrice(cur.pendPrice) + " | "
        else if bar_index - cur.pendConfBar > reclaimBarsInput
            evLabel(cur, bar_index, cur.side == 1 ? low : high, nmOf(cur, "BREAKDOWN", "BREAKOUT"), true, failColorInput, eventSize, "No reclaim within the deadline: range failed")
            if cur.side == 1
                accBad += 1
            else
                distBad += 1
            evFailed := true
            eventMsg += nmOf(cur, "Accumulation", "Distribution") + " range FAILED (no reclaim) | "
            cur.pendBreak := false
            closeRange := true
            failedRange := true
            closeTag := "BROKEN"

    // 4) SOS failure: back below the midline before LPS
    if not closeRange and cur.brokeOut and not cur.lpsSeen and bar_index > cur.breakBar and uC < 0.5
        evLabel(cur, bar_index, cur.side == 1 ? low : high, nmOf(cur, "SOS lost", "SOW lost"), true, failColorInput, eventSize, "Close back below the midline before LPS: breakout failed")
        if cur.side == 1
            accBad += 1
        else
            distBad += 1
        tgtMiss += removeOutcome(cur.id)
        evFailed := true
        eventMsg += nmOf(cur, "SOS", "SOW") + " FAILED — back below midline | "
        closeRange := true
        failedRange := true
        closeTag := "BREAKOUT LOST"

    // 5) SOS / SOW: breakout close with a quality leg (leg runs from the last climax-side swing)
    if not closeRange and not cur.brokeOut and not cur.pendBreak and legDir == cur.side and (cur.nTests > 0 or cur.sprung)
        [le, lp, ln] = legStats(legStartBar, bar_index)
        legEffNow  := le
        legPeakNow := lp
        legAtrNow  := safeDiv(math.abs(close - legStartPrice), atr, 0.0)
        bool effOk  = legEffNow >= sosEffInput
        bool lenOk  = legAtrNow >= sosLegAtrInput
        bool qualOk = sosLogicInput == "Effort AND leg length" ? (effOk and lenOk) : (effOk or lenOk)
        if uC > 1.0 + sosBufAtrInput * atr / h and qualOk
            float tCpos = cur.side == 1 ? cpos : 1.0 - cpos
            cur.sosScore := math.round(clamp01((legEffNow - 1.0) / 1.5) * 40.0 + clamp01(legAtrNow / 3.0) * 30.0 + tCpos * 30.0)
            cur.brokeOut := true
            cur.breakBar  := bar_index
            cur.sosEff  := legEffNow
            cur.stage   := 4
            string sosDet = str.tostring(cur.sosScore) + "/100 · leg eff ×" + str.tostring(legEffNow, "#.#") + " · " + str.tostring(legAtrNow, "#.#") + " ATR"
            evLabel(cur, bar_index, cur.side == 1 ? high : low, nmOf(cur, "SOS", "SOW") + (detailLblInput ? "\n" + sosDet : ""), false, rngCol, eventSize, nmOf(cur, "Sign of Strength", "Sign of Weakness") + "\n" + sosDet)
            addDivider(cur, bar_index, cur.sprung ? 3 : 2, rngCol)
            sigDetect := true
            eventLevel := cur.side == 1 ? cur.top : cur.bot
            entryScoreEv := cur.sosScore
            entryInfoEv := nmOf(cur, "SOS", "SOW") + " @ " + fmtPrice(close) + " · leg eff ×" + str.tostring(legEffNow, "#.#")
            if cur.side == 1
                evSos := true
            else
                evSow := true
            if cur.sprung
                if cur.side == 1
                    sprungBroke += 1
                else
                    utadBroke += 1
            cur.target := pOf(cur, 1.0 + projMultInput)
            if showTargetInput
                cur.tgtLn    := line.new(bar_index, cur.target, bar_index + projBarsInput, cur.target, xloc = xloc.bar_index, color = inkOf(rngCol), style = line.style_dashed, width = 2)
                cur.tgtTag := label.new(bar_index + projBarsInput, cur.target, "Target " + fmtPrice(cur.target), xloc = xloc.bar_index, style = label.style_label_left, color = eventBackdropInput ? LABEL_PLATE : TRANSP_COLOR, textcolor = inkOf(rngCol), size = eventSize, text_font_family = font.family_monospace)
            if math.abs(cur.target - close) >= tgtMarginAtrInput * atr
                array.push(oId, cur.id)
                array.push(oDir, cur.side)
                array.push(oTgt, cur.target)
                array.push(oInv, pOf(cur, 0.5))
                array.push(oBar, bar_index)
            else
                tgtPre += 1
            eventMsg += nmOf(cur, "SOS", "SOW") + " " + str.tostring(cur.sosScore) + "/100 at " + fmtPrice(close) + " → target " + fmtPrice(cur.target) + " | "

    // 6) LPS timeout
    if not closeRange and cur.brokeOut and not cur.lpsSeen and bar_index - cur.breakBar > lpsMaxBarsInput
        if cur.side == 1
            accOk += 1
        else
            distOk += 1
